```java
private void floodFill(int x, int y, int remainingMoves, int[][] visitedMoves,
        int startX, int startY, StatsComponent.MoveType moveType, UnitType unitType) {
    if (remainingMoves < 0) return;
    if (x < 0 || x >= gameMap.width || y < 0 || y >= gameMap.height) return;
    if (visitedMoves[x][y] >= remainingMoves) return;

    boolean isStart = (x == startX && y == startY);
    if (!isStart && !isWalkable(x, y, moveType, unitType)) return;

    visitedMoves[x][y] = remainingMoves;
    if (!isStart && getEntityAt(x, y, TypeComponent.Type.MARKER) == null) {
        entityFactory.createMovementMarker(x, y);
    }

    int next = remainingMoves - 1;
    floodFill(x + 1, y, next, visitedMoves, startX, startY, moveType, unitType);
    floodFill(x - 1, y, next, visitedMoves, startX, startY, moveType, unitType);
    floodFill(x, y + 1, next, visitedMoves, startX, startY, moveType, unitType);
    floodFill(x, y - 1, next, visitedMoves, startX, startY, moveType, unitType);
    floodFill(x + 1, y + 1, next, visitedMoves, startX, startY, moveType, unitType);
    floodFill(x - 1, y + 1, next, visitedMoves, startX, startY, moveType, unitType);
    floodFill(x + 1, y - 1, next, visitedMoves, startX, startY, moveType, unitType);
    floodFill(x - 1, y - 1, next, visitedMoves, startX, startY, moveType, unitType);
}
```

    ```java
    /**
     * Returns the movement cost (in x2-scaled units) for stepping from (fromX,fromY) to (toX,toY).
     * Rail-to-rail steps for LAND units cost 1 (half). All other steps cost 2 (normal).
     * Rail bonus does NOT apply in enemy territory.
     */
    private int getRailStepCost(int fromX, int fromY, int toX, int toY, StatsComponent.MoveType moveType, int unitOwner) {
        if (moveType == StatsComponent.MoveType.LAND
                && fromX >= 0 && fromX < gameMap.width && fromY >= 0 && fromY < gameMap.height
                && toX >= 0 && toX < gameMap.width && toY >= 0 && toY < gameMap.height
                && gameMap.rails[fromX][fromY]
                && gameMap.rails[toX][toY]) {
            // Check territory: no rail bonus in enemy territory
            // Use findControllingBase or equivalent to check if fromTile is enemy-owned
            // If the from-tile is owned by the opposing player, no bonus
            int[] fromTerritory = findControllingBase(fromX, fromY, unitOwner);
            // fromTerritory[0] == 1 means it's in unitOwner's territory, == 0 means neutral/enemy
            // We need to also check the to-tile isn't enemy territory
            int[] toTerritory = findControllingBase(toX, toY, unitOwner);
            // Rail bonus applies if NEITHER tile is in enemy territory
            // findControllingBase returns territory[0]==1 if tile is in owner's territory
            // For neutral territory, territory[0]==0 but no enemy owns it
            // We block bonus ONLY if the tile is in a DIFFERENT player's territory
            // Since findControllingBase checks against 'owner' parameter, territory[0]==1 means friendly
            // If territory[0]==0, it could be neutral OR enemy — need a more specific check
            // SIMPLER approach: just check if both tiles are NOT in enemy territory
            // The existing code at line 695 calls findControllingBase(x, y, owner) where owner = screen.getActiveLocalPlayer()
            // If findControllingBase returns [0,...] it means NOT in this owner's territory (could be neutral or enemy)
            // For rail purposes: apply bonus if both tiles are in friendly OR neutral territory
            // Block bonus ONLY if a tile is in the OTHER player's territory
            // To detect enemy territory specifically: call findControllingBase with the OTHER player's ID
            int enemyOwner = (unitOwner == 1) ? 2 : 1;
            int[] fromEnemyCheck = findControllingBase(fromX, fromY, enemyOwner);
            int[] toEnemyCheck = findControllingBase(toX, toY, enemyOwner);
            if (fromEnemyCheck[0] == 1 || toEnemyCheck[0] == 1) {
                return 2; // In enemy territory — no rail bonus
            }
            return 1; // Half cost (in x2 scaled units)
        }
        return 2; // Normal cost (in x2 scaled units)
    }
    ```
